ACME-MIB DEFINITIONS ::= BEGIN

-- acme - certificates kept, renewed and deployed by themselves: what each
-- node's own SNMPv3 agent
-- answers (1.0.182). Read only. Under netSnmpPlaypen (net-snmp's subtree for
-- objects of one's own), .7 - acme has no enterprise number of its own.
-- Every sample of https://NODE/metrics is in acmeMetricTable too - the
-- certificates the certificate manager keeps among them; they have a table of
-- their own as well (acmeManagedCertTable, under acmeCertificates). { acme 2 } and
-- { acme 4 } are not used (they were RADIUS and the tokens of otp).

IMPORTS
    MODULE-IDENTITY, OBJECT-TYPE, NOTIFICATION-TYPE,
    Integer32, Gauge32, Counter32, TimeTicks
        FROM SNMPv2-SMI
    DisplayString
        FROM SNMPv2-TC
    netSnmpPlaypen
        FROM NET-SNMP-MIB;

acme MODULE-IDENTITY
    LAST-UPDATED "202611010100Z"
    ORGANIZATION "acme"
    CONTACT-INFO "The administrators of the acme servers - Settings, Monitoring."
    DESCRIPTION  "acme, certificates kept, renewed and deployed by themselves:
                  the database, the alerts, the certificates of the pages, the
                  cluster and every metric of one node."
    REVISION     "202611010100Z"
    DESCRIPTION  "The certificates the certificate manager keeps: acmeManagedCertTable
                  and acmeManagedCounts under acmeCertificates."
    REVISION     "202611010000Z"
    DESCRIPTION  "acme: the RADIUS and token objects of otp taken out."
    REVISION     "202610300000Z"
    DESCRIPTION  "First: 1.0.182."
    ::= { netSnmpPlaypen 7 }

acmeNotifications OBJECT IDENTIFIER ::= { acme 0 }
acmeInfo          OBJECT IDENTIFIER ::= { acme 1 }
acmeDatabase      OBJECT IDENTIFIER ::= { acme 3 }
acmeAlerts        OBJECT IDENTIFIER ::= { acme 5 }
acmeCertificates  OBJECT IDENTIFIER ::= { acme 6 }
acmeCluster       OBJECT IDENTIFIER ::= { acme 7 }
acmeTrapObjects   OBJECT IDENTIFIER ::= { acme 8 }
acmeMetrics       OBJECT IDENTIFIER ::= { acme 9 }

-- The certificates the certificate manager keeps (Certificates on the pages,
-- acme --certs), switched on ones; the index is the certificate's id in acme,
-- the same while it is kept.

acmeManagedCertTable OBJECT-TYPE
    SYNTAX SEQUENCE OF AcmeManagedCertEntry MAX-ACCESS not-accessible STATUS current
    DESCRIPTION "The certificates acme keeps, renews and deploys." ::= { acmeCertificates 2 }
acmeManagedCertEntry OBJECT-TYPE
    SYNTAX AcmeManagedCertEntry MAX-ACCESS not-accessible STATUS current
    DESCRIPTION "One certificate." INDEX { acmeManagedCertIndex } ::= { acmeManagedCertTable 1 }
AcmeManagedCertEntry ::= SEQUENCE { acmeManagedCertIndex Integer32,
    acmeManagedCertName DisplayString, acmeManagedCertCN DisplayString,
    acmeManagedCertNotAfter Integer32, acmeManagedCertDaysLeft Integer32,
    acmeManagedCertState DisplayString, acmeManagedCertIssuer DisplayString,
    acmeManagedCertStateNum Integer32, acmeManagedCertAutoRenew Integer32,
    acmeManagedCertDeploysFailed Gauge32 }
acmeManagedCertIndex OBJECT-TYPE
    SYNTAX Integer32 (1..2147483647) MAX-ACCESS read-only STATUS current
    DESCRIPTION "The certificate's id in acme." ::= { acmeManagedCertEntry 1 }
acmeManagedCertName OBJECT-TYPE
    SYNTAX DisplayString MAX-ACCESS read-only STATUS current
    DESCRIPTION "Its name in acme." ::= { acmeManagedCertEntry 2 }
acmeManagedCertCN OBJECT-TYPE
    SYNTAX DisplayString MAX-ACCESS read-only STATUS current
    DESCRIPTION "Its common name (the first of its names)." ::= { acmeManagedCertEntry 3 }
acmeManagedCertNotAfter OBJECT-TYPE
    SYNTAX Integer32 MAX-ACCESS read-only STATUS current
    DESCRIPTION "When it ends, unix time; 0 when not issued yet." ::= { acmeManagedCertEntry 4 }
acmeManagedCertDaysLeft OBJECT-TYPE
    SYNTAX Integer32 UNITS "days" MAX-ACCESS read-only STATUS current
    DESCRIPTION "Days until it ends (below 0: ended); 0 when not issued yet."
    ::= { acmeManagedCertEntry 5 }
acmeManagedCertState OBJECT-TYPE
    SYNTAX DisplayString MAX-ACCESS read-only STATUS current
    DESCRIPTION "new, ok, due, renewing, failed, expired, pending, revoked or disabled."
    ::= { acmeManagedCertEntry 6 }
acmeManagedCertIssuer OBJECT-TYPE
    SYNTAX DisplayString MAX-ACCESS read-only STATUS current
    DESCRIPTION "Where it comes from (a source); empty for one imported."
    ::= { acmeManagedCertEntry 7 }
acmeManagedCertStateNum OBJECT-TYPE
    SYNTAX Integer32 (0..8) MAX-ACCESS read-only STATUS current
    DESCRIPTION "The state as a number: 0 new, 1 ok, 2 due, 3 renewing, 4 failed, 5 expired,
                 6 pending, 7 revoked, 8 disabled." ::= { acmeManagedCertEntry 8 }
acmeManagedCertAutoRenew OBJECT-TYPE
    SYNTAX Integer32 (0..1) MAX-ACCESS read-only STATUS current
    DESCRIPTION "1 when it is renewed by itself." ::= { acmeManagedCertEntry 9 }
acmeManagedCertDeploysFailed OBJECT-TYPE
    SYNTAX Gauge32 MAX-ACCESS read-only STATUS current
    DESCRIPTION "Its deployments whose last try failed." ::= { acmeManagedCertEntry 10 }
