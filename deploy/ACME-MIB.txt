-- acme - certificates kept, renewed and deployed by themselves: what each
-- node's own SNMPv3 agent
-- answers (1.0.182). Read only. Under netSnmpPlaypen (net-snmp's subtree for
-- objects of one's own), .7 - acme has no enterprise number of its own.
-- Every sample of https://NODE/metrics is in acmeMetricTable too - the
-- certificates the certificate manager keeps among them; they have a table of
-- their own as well (acmeManagedCertTable, under acmeCertificates). { acme 2 } and
-- { acme 4 } are not used (they were RADIUS and the tokens of otp).

IMPORTS
    MODULE-IDENTITY, OBJECT-TYPE, NOTIFICATION-TYPE,
    Integer32, Gauge32, Counter32, Counter64, TimeTicks
        FROM SNMPv2-SMI
    DisplayString, TEXTUAL-CONVENTION
        FROM SNMPv2-TC
    MODULE-COMPLIANCE, OBJECT-GROUP, NOTIFICATION-GROUP
        FROM SNMPv2-CONF
    CounterBasedGauge64
        FROM HCNUM-TC
    netSnmpPlaypen
        FROM NET-SNMP-MIB;

acme MODULE-IDENTITY
    LAST-UPDATED "202611161200Z"
    ORGANIZATION "acme"
    CONTACT-INFO "The administrators of the acme servers - Settings, Monitoring."
    DESCRIPTION  "acme, certificates kept, renewed and deployed by themselves:
                  the database, the alerts, the certificates of the pages, the
                  cluster and every metric of one node."
    REVISION     "202611161200Z"
    DESCRIPTION  "1.0.116: acmeAgentTable (acmeCertificates 9) - each target with acme-agent:
                  its name, mode, state, when it was last heard from, the days its
                  certificate has left, its version, whether a newer one is offered and
                  whether its policy blocks a deployment. Nothing before moved or changed."
    REVISION     "202611160000Z"
    DESCRIPTION  "1.0.115: the rows of acmeAlertTable, acmeCertTable, acmeNodeTable,
                  acmeMetricTable, acmeDiskTable, acmeRemoteBackupTable, acmeCrlTable and
                  acmeShipTable numbered by what names each row (the CRC-32 of its key, its

                  them in sorted order takes the next free one. acmeReplState: down(4). acmeBackupAge:
                  with no backup ever made, the time since acme started (it was 0)."
    REVISION     "202611150000Z"
    DESCRIPTION  "1.0.108: everything the pages show of a node, as objects of their own -
                  acmeSystem (the disks, the processor, the memory, the clock, SELinux, the

                  sent away and tried), acmeSignIn (the accounts, the lockouts, Windows and
                  certificate sign-ins, their CRLs), acmeServices (isemon and the syslog
                  servers, the mail queue, the directories, the SMS gateway), acmeDnsServer;
                  more of the cluster (the VIP address, the replication, the split, each
                  node's version and role), of the database (its size, the guard), of acme

                  alert as columns; the owner, last renewal, failures and serial of each

    DESCRIPTION  "1.0.78: acmeDeployTable, acmeTestTable, acmeModuleTable, acmeCaTable; the key
                  and the twin of a dual certificate and its failing tests in

    REVISION     "202611010100Z"
    DESCRIPTION  "The certificates the certificate manager keeps: acmeManagedCertTable

-- Textual conventions (1.0.108): every state a number, as a monitoring system wants it; the
-- objects of before keep their syntax.

AcmeTruth ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION "No or yes; off or on."
    SYNTAX INTEGER { no(0), yes(1) }

AcmeCertState ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION "The state of a certificate kept, as acmeManagedCertStateNum has it."
    SYNTAX INTEGER { new(0), ok(1), due(2), renewing(3), failed(4), expired(5), pending(6),
                     revoked(7), disabled(8), revokeFailed(9), deleting(10),
                     pendingApproval(11), declined(12), held(13) }

AcmeDeployStatus ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION "The state of a deployment (acmeDeployState as a number)."
    SYNTAX INTEGER { new(0), ok(1), changed(2), deploying(3), failed(4) }

AcmeAgentCondition ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION "The state of a target's acme-agent (acmeAgentState as a number): new - not
                 joined; answers - heard from lately; quiet - not heard from for ten minutes;
                 refused - it was refused; joining - a join under way; off - the target is
                 switched off."
    SYNTAX INTEGER { new(0), answers(1), quiet(2), refused(3), joining(4), off(5) }

AcmeTestResult ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION "How a test of a certificate went last (acmeTestState as a number): none - not
                 run yet; old - an older version is served; other - another certificate;
                 answer - a wrong answer; unreachable - the server did not answer."
    SYNTAX INTEGER { none(0), ok(1), warn(2), old(3), other(4), answer(5), unreachable(6) }

AcmeSeverity ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION "How bad a thing the alerts see is: ok - nothing; info; warn; bad."
    SYNTAX INTEGER { ok(0), info(1), warn(2), bad(3) }

AcmeResult ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION "How the last try went: none - never tried (or not set up); ok; failed."
    SYNTAX INTEGER { none(0), ok(1), failed(2) }

AcmeReplication ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION "The replication between two nodes: notReplicated - one database or one node;
                 healthy - both ways running; broken - a way stopped; lost - a slot lost, the
                 other node has to be made again from this one; down - replicated, and neither
                 the subscription nor the slot is there (1.0.115)."
    SYNTAX INTEGER { notReplicated(0), healthy(1), broken(2), lost(3), down(4) }

AcmeSplit ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION "Two replicating nodes that lost each other (docs/split-brain.md): inactive -
                 not a replicated pair, or nothing of it switched on; together; apart."
    SYNTAX INTEGER { inactive(0), together(1), apart(2) }

AcmeVrrpRole ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION "What keepalived said of the VIP address on a node last."
    SYNTAX INTEGER { none(0), master(1), backup(2), fault(3) }

AcmeSelinux ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION "SELinux on a node: unknown - not looked at yet; absent - no SELinux on the
                 machine; disabled; permissive; enforcing."
    SYNTAX INTEGER { unknown(0), absent(1), disabled(2), permissive(3), enforcing(4) }

AcmeTimeSync ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION "Whether the clock is kept in time (chrony, or timedatectl without it)."
    SYNTAX INTEGER { unknown(0), synced(1), notSynced(2) }
